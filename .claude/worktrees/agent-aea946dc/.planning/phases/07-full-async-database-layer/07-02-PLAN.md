    async def is_confirmed_empty_async(
        self, symbol: str, interval: str, start_date: date, end_date: date
    ) -> bool:
        """检查指定区间是否已被确认为空数据（异步版本）."""
        try:
            async with async_db_manager.session() as session:
                stmt = select(EmptyQueryResult).where(

                        EmptyQueryResult.interval == interval,
                        EmptyQueryResult.start_date <= start_date,

    ```python
    async def record_empty_result_async(
        self, symbol: str, interval: str, start_date: date, end_date: date
    ) -> None:
        """记录一个空查询结果（异步版本）."""
        try:
            async with async_db_manager.session() as session:
                stmt = select(EmptyQueryResult).where(
                    and_(
                        EmptyQueryResult.symbol == symbol,
                        EmptyQueryResult.interval == interval,
                        EmptyQueryResult.start_date <= start_date,
                        EmptyQueryResult.end_date >= end_date,
                    )
                )
                result = await session.execute(stmt)
                existing = result.scalar_one_or_none()

                if not existing:
                    record = EmptyQueryResult(
                        symbol=symbol,
                        interval=interval,
                        start_date=start_date,
                        end_date=end_date,
                    )
                    session.add(record)
                    await session.commit()
        except RuntimeError as exc:
            if "Database not initialized" in str(exc):
                logger.debug("数据库未初始化，跳过空查询缓存写入")
                return
            raise
    ```

  <files>src/galaxy_db_sync/database/services/kline_data_sync.py</files>
  <action>
    在 `KlineDataSyncService` 类中创建异步版本的分组方法：

    ```python
    async def _group_by_missing_ranges_async(
        self,
        completeness_results: dict,
        interval: str,
        full_start: str | None = None,
        full_end: str | None = None,
    ) -> dict[tuple[int, int], list[str]]:
        """按缺失区间对 symbol 进行分组（异步版本）."""
        from collections import defaultdict
        from datetime import date

        range_groups: dict[tuple[int, int], list[str]] = defaultdict(list)
        confirmed_missing_symbols = []

        for symbol, result in completeness_results.items():
            if result.is_complete:
                continue

            if not result.missing_ranges:
                continue

            # 异步记录缺失范围
            for range_start, range_end in result.missing_ranges:
                await missing_data_tracker.record_missing_range_async(
                    symbol=symbol,
                    interval=interval,
                    check_start=date.fromisoformat(str(full_start)) if full_start else range_start,
                    check_end=date.fromisoformat(str(full_end)) if full_end else range_end,
                    missing_start=range_start,
                    missing_end=range_end,
                )

            # 异步检查所有缺失范围是否都已确认
            all_confirmed = True
            for range_start, range_end in result.missing_ranges:
                if not await missing_data_tracker.is_range_confirmed_missing_async(
                    symbol, interval, range_start, range_end
                ):
                    all_confirmed = False
                    break

            if all_confirmed:
                confirmed_missing_symbols.append(symbol)
                continue

            # 按各自的缺失区间分组
            for range_start, range_end in result.missing_ranges:
                # 异步跳过已确认缺失的范围
                if await missing_data_tracker.is_range_confirmed_missing_async(
                    symbol, interval, range_start, range_end
                ):
                    continue

                begin_int = int(range_start.strftime("%Y%m%d"))
                end_int = int(range_end.strftime("%Y%m%d"))
                key = (begin_int, end_int)
                range_groups[key].append(symbol)

        if confirmed_missing_symbols:
            logger.info(
                f"跳过已确认缺失的 {len(confirmed_missing_symbols)} 只证券（如停牌期间）"
            )

        return dict(range_groups)
    ```
