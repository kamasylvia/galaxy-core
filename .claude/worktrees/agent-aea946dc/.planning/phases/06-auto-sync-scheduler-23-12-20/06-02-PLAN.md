    ```python
    """Sync state model for tracking synchronization history."""

    from datetime import datetime
    from sqlalchemy import Column, Integer, String, DateTime, Text, Index
    from galaxy_db_sync.database.models import Base


    class SyncState(Base):
        """Sync state tracking table.

        Tracks sync history to prevent duplicate syncs and allow retry tracking.
        """

        __tablename__ = "sync_state"

        id = Column(Integer, primary_key=True)
        sync_type = Column(String(50), nullable=False)  # 'kline_day', 'kline_week', 'calendar'
        period_identifier = Column(String(20), nullable=False)  # '2025-03-28', '2025-W12', '2026'
        sync_status = Column(String(20), nullable=False)  # 'pending', 'running', 'success', 'failed'
        started_at = Column(DateTime, nullable=True)
        completed_at = Column(DateTime, nullable=True)
        records_synced = Column(Integer, default=0)
        error_message = Column(Text, nullable=True)
        retry_count = Column(Integer, default=0)
        created_at = Column(DateTime, default=datetime.utcnow)
        updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

        __table_args__ = (
            Index('idx_sync_state_unique', 'sync_type', 'period_identifier', unique=True),
        )

        def __repr__(self) -> str:
            return f"<SyncState({self.sync_type}, {self.period_identifier}, {self.sync_status})>"
    ```

    ```python
    """Sync state manager for database operations."""

    import logging
    from datetime import datetime
    from sqlalchemy import select, update
    from sqlalchemy.dialects.postgresql import insert

    from galaxy_db_sync.database.core.async_connection import async_db_manager
    from galaxy_db_sync.database.models.sync_state import SyncState

    logger = logging.getLogger(__name__)


    class SyncStateManager:
        """Manages sync state records in database.

        Prevents duplicate syncs and tracks retry history.
        """

        async def is_synced(self, sync_type: str, period_identifier: str) -> bool:
            """Check if this sync was already completed successfully.

            Args:
                sync_type: e.g., 'kline_day', 'kline_week', 'calendar'
                period_identifier: e.g., '2025-03-28', '2025-W12', '2026'

            Returns:
                True if sync already succeeded, False otherwise
            """
            async with async_db_manager.session() as session:
                query = select(SyncState).where(
                    SyncState.sync_type == sync_type,
                    SyncState.period_identifier == period_identifier,
                    SyncState.sync_status == "success"
                )
                result = await session.execute(query)
                record = result.scalar_one_or_none()
                return record is not None

        async def mark_running(self, sync_type: str, period_identifier: str) -> None:
            """Mark sync as running (started).

            Creates new record or updates existing pending/failed record.
            """
            async with async_db_manager.session() as session:
                # Upsert: insert if not exists, update if exists
                stmt = insert(SyncState).values(
                    sync_type=sync_type,
                    period_identifier=period_identifier,
                    sync_status="running",
                    started_at=datetime.utcnow(),
                ).on_conflict_do_update(
                    constraint="idx_sync_state_unique",
                    set_={
                        "sync_status": "running",
                        "started_at": datetime.utcnow(),
                        "updated_at": datetime.utcnow(),
                    }
                )
                await session.execute(stmt)
            logger.info(f"Sync marked running: {sync_type} {period_identifier}")

        async def mark_success(
            self,
            sync_type: str,
            period_identifier: str,
            records_synced: int = 0
        ) -> None:
            """Mark sync as successfully completed.

            Args:
                sync_type: Sync type identifier
                period_identifier: Period identifier
                records_synced: Number of records synced
            """
            async with async_db_manager.session() as session:
                stmt = update(SyncState).where(
                    SyncState.sync_type == sync_type,
                    SyncState.period_identifier == period_identifier,
                ).values(
                    sync_status="success",
                    completed_at=datetime.utcnow(),
                    records_synced=records_synced,
                    updated_at=datetime.utcnow(),
                )
                await session.execute(stmt)
            logger.info(f"Sync succeeded: {sync_type} {period_identifier}, {records_synced} records")

        async def mark_failed(
            self,
            sync_type: str,
            period_identifier: str,
            error_message: str
        ) -> None:
            """Mark sync as failed.

            Args:
                sync_type: Sync type identifier
                period_identifier: Period identifier
                error_message: Error description
            """
            async with async_db_manager.session() as session:
                # Get current retry count
                query = select(SyncState.retry_count).where(
                    SyncState.sync_type == sync_type,
                    SyncState.period_identifier == period_identifier,
                )
                result = await session.execute(query)
                current_retry = result.scalar_one_or_none() or 0

                stmt = update(SyncState).where(
                    SyncState.sync_type == sync_type,
                    SyncState.period_identifier == period_identifier,
                ).values(
                    sync_status="failed",
                    error_message=error_message,
                    retry_count=current_retry + 1,
                    updated_at=datetime.utcnow(),
                )
                await session.execute(stmt)
            logger.error(f"Sync failed: {sync_type} {period_identifier} - {error_message}")

        async def get_retry_count(self, sync_type: str, period_identifier: str) -> int:
            """Get current retry count for a sync."""
            async with async_db_manager.session() as session:
                query = select(SyncState.retry_count).where(
                    SyncState.sync_type == sync_type,
                    SyncState.period_identifier == period_identifier,
                )
                result = await session.execute(query)
                return result.scalar_one_or_none() or 0
    ```

    ```python
    """Retry handler with exponential backoff.

    Per SCHED-03: Retry 3 times with delays 5min, 15min, 60min.
    """

    import asyncio
    import logging

    from galaxy_db_sync.shared_kernel.utils.config_loader import get_section

    logger = logging.getLogger(__name__)


    class RetryPolicy:
        """Retry configuration per SCHED-03."""

        # Default values from SCHED-03
        MAX_RETRIES: int = 3
        RETRY_DELAYS: list[int] = [5 * 60, 15 * 60, 60 * 60]  # 5min, 15min, 60min in seconds

        @classmethod
        def from_config(cls) -> "RetryPolicy":
            """Load retry policy from config."""
            config = get_section("scheduler")
            cls.MAX_RETRIES = config.get("max_retries", 3)
            delays_minutes = config.get("retry_delays_minutes", [5, 15, 60])
            cls.RETRY_DELAYS = [d * 60 for d in delays_minutes]
            return cls()


    async def sync_with_retry(
        sync_func: callable,
        sync_name: str,
        *args,
        **kwargs
    ) -> bool:
        """Execute sync with retry policy per SCHED-03.

        Args:
            sync_func: Async function to execute
            sync_name: Human-readable name for logging
            *args: Arguments to pass to sync_func
            **kwargs: Keyword arguments to pass to sync_func

        Returns:
            True if sync succeeded, False if all retries exhausted
        """
        RetryPolicy.from_config()

        for attempt, delay in enumerate(RetryPolicy.RETRY_DELAYS, start=1):
            try:
                await sync_func(*args, **kwargs)
                logger.info(f"{sync_name} sync succeeded on attempt {attempt}")
                return True
            except Exception as e:
                logger.error(f"{sync_name} sync failed (attempt {attempt}): {e}")
                if attempt < RetryPolicy.MAX_RETRIES:
                    logger.info(f"Retrying in {delay // 60} minutes...")
                    await asyncio.sleep(delay)

        logger.error(f"{sync_name} sync failed after {RetryPolicy.MAX_RETRIES} attempts")
        return False
    ```
