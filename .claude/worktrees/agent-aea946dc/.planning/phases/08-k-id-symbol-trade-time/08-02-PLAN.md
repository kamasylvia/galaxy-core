```python
#!/usr/bin/env python3
"""K 线表主键迁移脚本：将自增 id 改为 (symbol, trade_time) 组合主键.

执行流程：
1. 备份每个表到 backup 表
2. 验证备份行数
3. 删除旧表（级联删除 hypertable）
4. 创建新表（使用新模型，组合主键）
5. 从备份恢复数据（排除 id 列）
6. 验证数据行数
7. 重建 hypertable
8. 可选：删除备份表

Usage:
    python scripts/migrate_pk_to_symbol_trade_time.py [--dry-run] [--keep-backup]
"""

import asyncio
import argparse
import logging
from datetime import datetime
from sqlalchemy import text

from galaxy_db_sync.database.core.async_connection import async_db_manager
from galaxy_db_sync.database.core.timescale import create_hypertable

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# 13 个 K 线表，按大小从小到大排序（min 表数据量大，优先小表）
**KLINE_TABLES 按从小到大排序（year -> min1），以便小表先迁移，减少长时间锁定风险。**

KLINE_TABLES = [
    "kline_year",      # 数据量最小，优先迁移
    "kline_season",
    "kline_month",
    "kline_week",
    "kline_day",
    "kline_min120",
    "kline_min60",
    "kline_min30",
    "kline_min15",
    "kline_min10",
    "kline_min5",
    "kline_min3",
    "kline_min1",      # 数据量最大，最后迁移
]

async def get_table_row_count(session, table_name: str) -> int:
    """获取表的行数."""
    result = await session.execute(text(f'SELECT count(*) FROM "{table_name}"'))
    return result.scalar()

async def create_backup_table(session, table_name: str, backup_name: str) -> int:
    """创建备份表并返回行数."""
    logger.info(f"创建备份表: {backup_name}")
    await session.execute(text(f'CREATE TABLE "{backup_name}" AS SELECT * FROM "{table_name}"'))
    await session.commit()
    count = await get_table_row_count(session, backup_name)
    logger.info(f"备份表 {backup_name} 创建完成，行数: {count}")
    return count

async def drop_old_table(session, table_name: str) -> None:
    """删除旧表（级联删除 hypertable）."""
    logger.info(f"删除旧表: {table_name}")
    await session.execute(text(f'DROP TABLE "{table_name}" CASCADE'))
    await session.commit()

async def create_new_table(session, table_name: str) -> None:
    """创建新表（组合主键，无 id 列）."""
    logger.info(f"创建新表: {table_name}")
    # 新表结构：组合主键 (symbol, trade_time)
    create_sql = f"""
    CREATE TABLE "{table_name}" (
        symbol VARCHAR(20) NOT NULL,
        trade_time TIMESTAMP NOT NULL,
        open DOUBLE PRECISION,
        high DOUBLE PRECISION,
        low DOUBLE PRECISION,
        close DOUBLE PRECISION,
        volume BIGINT,
        amount DOUBLE PRECISION,
        security_type VARCHAR(20),
        created_at TIMESTAMP DEFAULT NOW(),
        updated_at TIMESTAMP DEFAULT NOW(),
        PRIMARY KEY (symbol, trade_time)
    );
    """
    await session.execute(text(create_sql))
    await session.commit()

async def restore_data(session, table_name: str, backup_name: str) -> int:
    """从备份表恢复数据."""
    logger.info(f"从备份恢复数据: {backup_name} -> {table_name}")
    insert_sql = f"""
    INSERT INTO "{table_name}" (symbol, trade_time, open, high, low, close, volume, amount, security_type, created_at, updated_at)
    SELECT symbol, trade_time, open, high, low, close, volume, amount, security_type, created_at, updated_at
    FROM "{backup_name}";
    """
    result = await session.execute(text(insert_sql))
    await session.commit()
    count = result.rowcount
    logger.info(f"恢复完成，插入行数: {count}")
    return count

async def create_hypertable_for_table(session, table_name: str) -> bool:
    """为新表创建 hypertable."""
    logger.info(f"创建 hypertable: {table_name}")
    # 使用 timescale.py 中的 create_hypertable 函数
    return await create_hypertable(session, table_name, time_column="trade_time")

async def migrate_table(session, table_name: str, timestamp: str, dry_run: bool = False) -> dict:
    """迁移单个表."""
    backup_name = f"{table_name}_backup_{timestamp}"
    result = {
        "table": table_name,
        "backup": backup_name,
        "original_count": 0,
        "backup_count": 0,
        "restored_count": 0,
        "success": False,
    }
    
    try:
        # 1. 获取原始表行数
        result["original_count"] = await get_table_row_count(session, table_name)
        logger.info(f"表 {table_name} 原始行数: {result['original_count']}")
        
        if dry_run:
            logger.info(f"[DRY RUN] 将迁移表: {table_name}")
            result["success"] = True
            return result
        
        # 2. 创建备份表
        result["backup_count"] = await create_backup_table(session, table_name, backup_name)
        
        # 3. 验证备份行数
        if result["original_count"] != result["backup_count"]:
            raise ValueError(f"备份行数不匹配: 原始 {result['original_count']}, 备份 {result['backup_count']}")
        
        # 4. 删除旧表
        await drop_old_table(session, table_name)
        
        # 5. 创建新表
        await create_new_table(session, table_name)
        
        # 6. 恢复数据
        result["restored_count"] = await restore_data(session, table_name, backup_name)
        
        # 7. 验证数据行数
        if result["original_count"] != result["restored_count"]:
            raise ValueError(f"恢复行数不匹配: 原始 {result['original_count']}, 恢复 {result['restored_count']}")
        
        # 8. 创建 hypertable
        hypertable_created = await create_hypertable_for_table(session, table_name)
        if not hypertable_created:
            logger.warning(f"创建 hypertable 可能失败: {table_name}")
        
        result["success"] = True
        logger.info(f"表 {table_name} 迁移成功")
        
    except Exception as e:
        logger.error(f"表 {table_name} 迁移失败: {e}")
        result["error"] = str(e)
        raise
    
    return result

async def migrate_all_tables(dry_run: bool = False, keep_backup: bool = False) -> list:
    """迁移所有 K 线表."""
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    results = []
    
    async with async_db_manager.session() as session:
        for table_name in KLINE_TABLES:
            try:
                result = await migrate_table(session, table_name, timestamp, dry_run)
                results.append(result)
            except Exception as e:
                logger.error(f"迁移中断: {e}")
                break
    
    # 汇总结果
    success_count = sum(1 for r in results if r.get("success"))
    logger.info(f"迁移完成: {success_count}/{len(KLINE_TABLES)} 个表成功")
    
    return results

def main():
    parser = argparse.ArgumentParser(description="K 线表主键迁移脚本")
    parser.add_argument("--dry-run", action="store_true", help="仅打印将要执行的操作，不实际执行")
    parser.add_argument("--keep-backup", action="store_true", help="保留备份表")
    args = parser.parse_args()
    
    asyncio.run(migrate_all_tables(dry_run=args.dry_run, keep_backup=args.keep_backup))

if __name__ == "__main__":
    main()
```

4. 支持 --dry-run 模式
5. 每步都有行数验证
6. 使用 CASCADE 删除旧表确保 hypertable 也被删除
7. 恢复数据时排除 id 列（新表无 id 列）

```python
#!/usr/bin/env python3
"""K 线表主键迁移回滚脚本：从备份表恢复原始表结构.

执行流程：
1. 查找指定表的备份表
2. 删除当前新表
3. 从备份表恢复原始表结构（包含 id 列）
4. 重建 hypertable
5. 可选：删除备份表

Usage:
    python scripts/rollback_pk_migration.py [--timestamp YYYYMMDD_HHMMSS] [--table kline_day] [--all]
"""

import asyncio
import argparse
import logging
from datetime import datetime
from sqlalchemy import text

from galaxy_db_sync.database.core.async_connection import async_db_manager
from galaxy_db_sync.database.core.timescale import create_hypertable

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

KLINE_TABLES = [
    "kline_min1", "kline_min3", "kline_min5", "kline_min10",
    "kline_min15", "kline_min30", "kline_min60", "kline_min120",
    "kline_day", "kline_week", "kline_month", "kline_season", "kline_year"
]

async def find_backup_tables(session, timestamp: str = None) -> dict:
    """查找备份表."""
    if timestamp:
        pattern = f"%_backup_{timestamp}"
    else:
        pattern = "%_backup_%"
    
    result = await session.execute(text("""
        SELECT table_name FROM information_schema.tables 
        WHERE table_schema = 'public' AND table_name LIKE :pattern
    """), {"pattern": pattern})
    
    backups = {}
    for row in result.fetchall():
        backup_name = row[0]
        # 从备份表名解析原表名 (e.g., kline_day_backup_20240423_120000 -> kline_day)
        if "_backup_" in backup_name:
            original = backup_name.split("_backup_")[0]
            backups[original] = backup_name
    
    return backups

async def rollback_table(session, table_name: str, backup_name: str) -> bool:
    """回滚单个表."""
    logger.info(f"回滚表: {table_name} <- {backup_name}")
    
    try:
        # 1. 删除当前表（如果存在）
        await session.execute(text(f'DROP TABLE IF EXISTS "{table_name}" CASCADE'))
        await session.commit()
        logger.info(f"删除当前表: {table_name}")
        
        # 2. 从备份表恢复（保留备份表结构）
        # 重命名备份表为原表名
        await session.execute(text(f'ALTER TABLE "{backup_name}" RENAME TO "{table_name}"'))
        await session.commit()
        logger.info(f"重命名备份表: {backup_name} -> {table_name}")
        
        # 3. 检查并重建 hypertable
        # 备份表已经是普通表，需要重新创建为 hypertable
        # 但由于备份表保留了原始结构，我们需要重新创建 hypertable
        # 实际上备份表是普通表，需要转换
        await session.execute(text(f"""
            SELECT create_hypertable('{table_name}', 'trade_time', if_not_exists => true);
        """))
        await session.commit()
        logger.info(f"重建 hypertable: {table_name}")
        
        return True
        
    except Exception as e:
        logger.error(f"回滚表 {table_name} 失败: {e}")
        await session.rollback()
        return False

async def rollback_all(timestamp: str = None, tables: list = None) -> dict:
    """回滚所有或指定表."""
    async with async_db_manager.session() as session:
        backups = await find_backup_tables(session, timestamp)
        
        if not backups:
            logger.warning("未找到备份表")
            return {}
        
        results = {}
        
        if tables:
            # 仅回滚指定表
            for table_name in tables:
                if table_name in backups:
                    success = await rollback_table(session, table_name, backups[table_name])
                    results[table_name] = success
                else:
                    logger.warning(f"未找到备份表: {table_name}")
                    results[table_name] = False
        else:
            # 回滚所有找到备份的表
            for table_name, backup_name in backups.items():
                success = await rollback_table(session, table_name, backup_name)
                results[table_name] = success
        
        return results

def main():
    parser = argparse.ArgumentParser(description="K 线表主键迁移回滚脚本")
    parser.add_argument("--timestamp", help="备份时间戳 (YYYYMMDD_HHMMSS)")
    parser.add_argument("--table", action="append", help="指定要回滚的表（可多次使用）")
    parser.add_argument("--all", action="store_true", help="回滚所有找到备份的表")
    args = parser.parse_args()
    
    if not args.all and not args.table:
        parser.error("请指定 --all 或 --table")
    
    tables = args.table if args.table else None
    results = asyncio.run(rollback_all(timestamp=args.timestamp, tables=tables))
    
    # 打印结果
    success_count = sum(1 for v in results.values() if v)
    logger.info(f"回滚完成: {success_count}/{len(results)} 个表成功")

if __name__ == "__main__":
    main()
```

3. 回滚逻辑：删除新表 -> 重命名备份表 -> 重建 hypertable
4. 保持备份表结构（包含 id 列）
